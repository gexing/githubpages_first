快速开始

只需几行代码即可连接数据库并执行查询。

import { QuantumDB, Model, Field } from 'quantumdb'; // 1. 初始化连接 const db = new QuantumDB({ dialect: 'postgresql', url: 'postgres://user:pass@localhost:5432/mydb', pool: { max: 20, min: 5 }, }); // 2. 定义模型 class User extends Model { id: number; name: string; email: string; } // 3. 执行查询 const users = await db.select(User) .where({ active: true }) .orderBy('name', 'asc') .limit(10) .execute();
const { QuantumDB } = require('quantumdb'); // 初始化 const db = new QuantumDB({ dialect: 'mysql', host: 'localhost', database: 'myapp', }); // 原生查询 const result = await db.raw( 'SELECT * FROM users WHERE age > ?', [18] ); console.log(result.rows);
# quantumdb.config.yaml dialect: postgresql connection: host: db.example.com port: 5432 database: production_db ssl: true pool: max: 50 idle_timeout: 30000 migrations: auto_run: true directory: ./db/migrations logging: level: info slow_query_threshold: 200 # ms
